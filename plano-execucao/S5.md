## Issue #6 — Metas SMART + mural

model Discussao {
  id       String   @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  ptsId    String   @db.Uuid
  autorId  String   @db.Uuid
  texto    String
  criadaEm DateTime @default(now()) @db.Timestamptz(3)

  pts   Pts     @relation(fields: [ptsId], references: [id], onDelete: Restrict)
  autor Usuario @relation("discussoes", fields: [autorId], references: [id], onDelete: Restrict)
